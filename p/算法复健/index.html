Random rand=new Random();

    private int partition(int[]nums,int left,int right)
    {
        int i=left+rand.nextInt(right-left+1);
        int pivot=nums[i];
        swap(nums,i,left);

        i=left+1;
        int j=right;
        while(true)
        {
            while(i<=j&&nums[i]<pivot)
            {
                i++;
            }
            while(i<=j&&nums[j]>pivot)
            {
                j--;
            }
            if(i>=j)
            {
                break;
            }
            swap(nums,i,j);
            i++;
            j--;
        }
        swap(nums,left,j);
        return j;
    }

    private void swap(int[]nums,int i,int j)
    {
        int temp=nums[i];
        nums[i]=nums[j];
        nums[j]=temp;
    }
